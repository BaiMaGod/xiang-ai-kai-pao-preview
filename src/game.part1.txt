const qs = new URLSearchParams(location.search);
const TEST_MODE = qs.get('test') === '1';
const el = id => document.getElementById(id);
const dom = {
  game: el('game'), hud: el('hud'), hpFill: el('hp-fill'), hpText: el('hp-text'),
  xpFill: el('xp-fill'), level: el('level-text'), timer: el('timer'), aiFill: el('ai-fill'), aiText: el('ai-text'),
  status: el('status-strip'), toast: el('toast'), modal: el('modal'), kicker: el('modal-kicker'), title: el('modal-title'),
  desc: el('modal-desc'), choices: el('choices'), start: el('start-screen'), end: el('end-screen'), startBtn: el('start-btn'),
  restartBtn: el('restart-btn'), endTitle: el('end-title'), endKicker: el('end-kicker'), endStats: el('end-stats'),
  joystick: el('joystick'), stick: el('stick'), skillBtn: el('skill-btn'), skillCd: el('skill-cd'), debug: el('debug')
};

window.addEventListener('error', e => { document.body.dataset.error = String(e.message || e.error || 'unknown'); });
window.addEventListener('unhandledrejection', e => { document.body.dataset.error = String(e.reason || 'unhandled rejection'); });

class Game {
  constructor() {
    this.model = new RunModel({ test: TEST_MODE });
    this.state = 'MENU';
    this.runPhase = 'NORMAL';
    this.modalState = 'NONE';
    this.paused = true;
    this.keys = new Set();
    this.moveInput = new THREE.Vector2();
    this.joyInput = new THREE.Vector2();
    this.enemyPool = [];
    this.projectilePool = [];
    this.enemyShotPool = [];
    this.pickupPool = [];
    this.effects = [];
    this.pendingLevels = 0;
    this.spawnSuppressed = false;
    this.lastTime = performance.now();
    this.toastTimer = null;
    this.stats = { totalDamage: 0, blockedHits: 0, aiFails: 0 };
    this.setupThree();
    this.buildWorld();
    this.bindInput();
    this.loop = this.loop.bind(this);
    requestAnimationFrame(this.loop);
    document.body.dataset.render = 'ok';
  }

  setupThree() {
    this.scene = new THREE.Scene();
    this.scene.background = new THREE.Color(0xbfd8d2);
    this.scene.fog = new THREE.Fog(0xbfd8d2, 22, 47);
    this.camera = new THREE.PerspectiveCamera(52, innerWidth / innerHeight, 0.1, 100);
    this.camera.position.set(8, 13, 11);
    this.renderer = new THREE.WebGLRenderer({ antialias: !TEST_MODE, powerPreference: 'high-performance' });
    this.renderer.setPixelRatio(Math.min(devicePixelRatio, 1.6));
    this.renderer.setSize(innerWidth, innerHeight);
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
    dom.game.appendChild(this.renderer.domElement);
    const hemi = new THREE.HemisphereLight(0xffffff, 0x53645c, 2.4);
    this.scene.add(hemi);
    const sun = new THREE.DirectionalLight(0xffffff, 2.2);
    sun.position.set(8, 16, 10);
    this.scene.add(sun);
    addEventListener('resize', () => {
      this.camera.aspect = innerWidth / innerHeight;
      this.camera.updateProjectionMatrix();
      this.renderer.setSize(innerWidth, innerHeight);
    });
  }

  mat(color, roughness=.75) { return new THREE.MeshStandardMaterial({ color, roughness, metalness:.12 }); }

  buildWorld() {
    this.staticGroup = new THREE.Group();
    this.scene.add(this.staticGroup);
    const ground = new THREE.Mesh(new THREE.PlaneGeometry(50,50), this.mat(0x9fb9a8,1));
    ground.rotation.x = -Math.PI/2;
    this.staticGroup.add(ground);
    const plaza = new THREE.Mesh(new THREE.PlaneGeometry(19,19), this.mat(0xc8c5b7,1));
    plaza.rotation.x = -Math.PI/2; plaza.position.y=.01;
    this.staticGroup.add(plaza);
    const roadMat = this.mat(0x697279,1);
    const road1 = new THREE.Mesh(new THREE.PlaneGeometry(50,6), roadMat); road1.rotation.x=-Math.PI/2; road1.position.y=.015;
    const road2 = road1.clone(); road2.rotation.z=Math.PI/2;
    this.staticGroup.add(road1,road2);
    for (const [x,z,w,d,h,c] of [
      [-18,-17,9,8,5,0xd9b37c],[17,-17,9,8,6,0xe5c58d],[-18,17,10,7,7,0xb7c7d3],[17,17,10,7,5,0xe0a9a2]
    ]) {
      const b=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),this.mat(c,.95)); b.position.set(x,h/2,z); this.staticGroup.add(b);
      for(let i=-1;i<=1;i++){ const win=new THREE.Mesh(new THREE.BoxGeometry(.9,.7,.06),this.mat(0x7ec5d6,.4)); win.position.set(x+i*2.2,Math.min(h-1.3,2.3),z+(z>0?-d/2-.04:d/2+.04)); this.staticGroup.add(win); }
    }
    for (const [x,z] of [[-9,-8],[10,-7],[-8,9],[8,8]]) {
      const planter=new THREE.Mesh(new THREE.BoxGeometry(4,.45,1.6),this.mat(0x786b55,1)); planter.position.set(x,.22,z); this.staticGroup.add(planter);
      for(let i=-1;i<=1;i++){const bush=new THREE.Mesh(new THREE.SphereGeometry(.55,8,6),this.mat(0x4f8254,1)); bush.position.set(x+i*1.05,.7,z); this.staticGroup.add(bush);}
    }
    const kiosk=new THREE.Mesh(new THREE.BoxGeometry(3,2,1.2),this.mat(0x6f8b9d)); kiosk.position.set(10,1,-12); this.staticGroup.add(kiosk);
  }

  createPlayer() {
    const g = new THREE.Group();
    const body = new THREE.Mesh(new THREE.CylinderGeometry(.34,.42,.85,10),this.mat(0x354b65)); body.position.y=.68;
    const head = new THREE.Mesh(new THREE.SphereGeometry(.27,12,8),this.mat(0xe3b98e)); head.position.y=1.28;
    const helmet = new THREE.Mesh(new THREE.SphereGeometry(.285,12,6,0,Math.PI*2,0,Math.PI/2),this.mat(0xf0ca46)); helmet.position.y=1.31;
    const gun = new THREE.Mesh(new THREE.BoxGeometry(.16,.14,.78),this.mat(0x3b3e43,.45)); gun.position.set(.42,.82,.18); gun.rotation.y=.05;
    g.add(body,head,helmet,gun); g.position.set(0,0,0); this.scene.add(g);
    return { mesh:g, hp:100, maxHp:100, invuln:0, speed:5.5, skillCd:0, radius:.45, gunLevel:1, ammoPrinter:0, armor:0, magnet:1.55, rpgLevel:0, empLevel:0, gunCd:0, rpgCd:0, empCd:0, evolved:false };
  }

    this.model.reset();
    this.stats = { totalDamage: 0, blockedHits: 0, aiFails: 0 };
